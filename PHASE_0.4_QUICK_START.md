## Use in Your Code

### Option 1: Through StepSequencingService (Recommended)

```python
from api.services.step_sequencing_service import StepSequencingService
from api.models.teaching_step import Subject

# Service automatically initializes all three experts
service = StepSequencingService(kg_service, model_gateway)

# Generate a teaching plan (experts automatically route)
plan = await service.generate_teaching_plan(
    problem="Solve 2x + 5 = 13",
    subject=Subject.MATHEMATICS,
    grade_level=10
)

# Plan has all steps generated by MathExpertService
print(f"First step: {plan.first_step.title}")
```

### Option 2: Direct Expert Usage

```python
from api.services.math_expert_service import MathExpertService

expert = MathExpertService()

# Analyze problem
analysis = await expert.analyze_problem("Solve x^2 + 5x + 6 = 0")
print(f"Problem type: {analysis.problem_type}")  # QUADRATIC_EQUATION

# Generate steps
steps = await expert.generate_steps(analysis)
print(f"Generated {len(steps)} steps")

# Evaluate student response
result = await expert.evaluate_response(
    question="Solve x^2 + 5x + 6 = 0",
    student_response="x = -2, x = -3",
    context={"expected_answer": "x = -2, x = -3"}
)
print(f"Correct: {result.is_correct}")
print(f"Feedback: {result.feedback_message}")
```

## Example: Complete Workflow

```python
from api.services.math_expert_service import MathExpertService
from api.models.teaching_step import StepType

# Create expert
expert = MathExpertService()

# 1. Analyze the problem
problem = "Solve 2x + 5 = 13"
analysis = await expert.analyze_problem(problem)
print(f"Type: {analysis.problem_type}")           # LINEAR_EQUATION
print(f"Difficulty: {analysis.difficulty}")       # 1 (easy)
print(f"Concepts: {analysis.concepts}")           # ['linear equations', ...]

# 2. Generate teaching steps
steps = await expert.generate_steps(analysis)
print(f"Generated {len(steps)} steps")

# Show each step
for step in steps:
    print(f"Step {step.step_number}: {step.title}")
    print(f"  Type: {step.type}")
    print(f"  Description: {step.description}")

# 3. Student starts first step (step with interaction)
step_with_question = next(s for s in steps if s.interaction)
print(f"\nQuestion: {step_with_question.interaction.question_text}")

# 4. Student responds
student_response = "x = 4"

# 5. Evaluate response
result = await expert.evaluate_response(
    question=step_with_question.interaction.question_text,
    student_response=student_response,
    context={
        "expected_answer": step_with_question.interaction.expected_answer,
        "problem_type": analysis.problem_type.value
    }
)

# 6. Get feedback
if result.is_correct:
    print(f"✓ Correct! {result.feedback_message}")
else:
    print(f"✗ Not quite. {result.feedback_message}")
    if result.should_offer_hint:
        print(f"Hint: {result.hint_text}")

# 7. Route to next step
next_step_id = result.next_step_id
next_step = next(s for s in steps if s.id == next_step_id)
print(f"\nNext: {next_step.title}")
```

## Common Patterns

### Pattern 1: Generate and Return Full Plan

```python
async def get_teaching_plan(problem: str, subject: str) -> dict:
    service = StepSequencingService()
    plan = await service.generate_teaching_plan(
        problem=problem,
        subject=Subject(subject),
        grade_level=10
    )
    return {
        "plan_id": plan.id,
        "steps": [step.dict() for step in plan.steps],
        "first_step_id": plan.first_step.id
    }
```

### Pattern 2: Evaluate and Route

```python
async def evaluate_student_response(
    plan_id: str, step_id: str, response: str
) -> dict:
    # Get plan and step from database
    plan = await db.get_plan(plan_id)
    step = plan.get_step_by_id(step_id)
    
    # Evaluate
    service = StepSequencingService()
    result = await service.evaluate_step_response(
        step=step,
        student_response=response
    )
    
    # Get next step
    next_step = service.get_next_step(plan, step_id, result)
    
    return {
        "is_correct": result.is_correct,
        "feedback": result.feedback_message,
        "next_step": next_step.dict() if next_step else None
    }
```

### Pattern 3: Stream Steps One-by-One

```python
async def stream_teaching_plan(problem: str, subject: str):
    service = StepSequencingService()
    plan = await service.generate_teaching_plan(
        problem=problem,
        subject=Subject(subject),
        grade_level=10
    )
    
    # Stream each step to client
    for step in plan.steps:
        yield {
            "step_number": step.step_number,
            "title": step.title,
            "description": step.description,
            "content": step.dict()
        }
        await asyncio.sleep(0.1)  # Small delay for UX
```

## Debugging Tips

### See What Expert Detected

```python
expert = MathExpertService()
analysis = await expert.analyze_problem("Your problem")
print(f"Detected type: {analysis.problem_type}")
print(f"Difficulty: {analysis.difficulty}")
print(f"Solution method: {analysis.solution_method}")
```

### See Generated Steps Structure

```python
steps = await expert.generate_steps(analysis)
for i, step in enumerate(steps):
    print(f"\nStep {i+1}: {step.type.value}")
    print(f"  ID: {step.id}")
    print(f"  Title: {step.title}")
    if step.interaction:
        print(f"  Question: {step.interaction.question_text}")
    print(f"  Next: {step.next_step_id}")
    for rule in step.branching_rules:
        print(f"  Branch: {rule.condition} → {rule.target_step_id}")
```

## Performance Tips

1. **Reuse Expert Instances**
   ```python
   # Good: Create once, reuse
   math_expert = MathExpertService()
   
   # Call multiple times
   analysis1 = await math_expert.analyze_problem(p1)
   analysis2 = await math_expert.analyze_problem(p2)
   ```

2. **Cache Analysis Results**
   ```python
   # Don't re-analyze same problem
   analysis = await expert.analyze_problem(problem)
   steps = await expert.generate_steps(analysis)  # Reuse analysis
   ```

3. **Stream Long Step Lists**
   ```python
   # Instead of waiting for all steps, stream them
   async for step in stream_teaching_plan(problem):
       yield step
   ```

## Troubleshooting

### Problem: Expert not initialized
```
AttributeError: 'StepSequencingService' has no attribute 'math_expert'
```
**Solution**: Make sure you're using the updated `StepSequencingService`

### Problem: Wrong problem type detected
```
analysis.problem_type == GEOMETRY (should be ALGEBRA)
```
**Solution**: Check if problem text is clear. Add more keywords.

### Problem: No steps generated
```
steps = await expert.generate_steps(analysis)
# Returns empty list
```
**Solution**: Check analysis is not None. Verify problem type is valid.
